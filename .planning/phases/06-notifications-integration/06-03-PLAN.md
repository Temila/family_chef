<objective>
Wire the Feishu delivery layer and the status-change timestamp writes so the badge from Plan 02 actually flips and chefs receive the right card. This plan adds (1) a dedicated `send_wish_notification()` card method, (2) a `WishNotificationService` that resolves recipients and isolates failures, and (3) all 5 Phase 5 hook locations in `wish_service.py` plus the `submit_wish` fan-out and the atomic claim timestamp. NOTIF-05 + NOTIF-06 are fully satisfied; the badge lifecycle from NOTIF-03 becomes observable because every status change now writes `last_status_change_at`.

Purpose: Phase 7 frontend can rely on a stable backend — claim/advance/reject/cancel edits the timestamp, new/claimed-edit/claimed-cancel sends the right Feishu card, and a Feishu failure never bubbles up to the wish HTTP response.

Output: A new `notification_service.py`, edits to `feishu.py` + `wish_service.py`, and a comprehensive test file proving targeting, isolation, and the claim-timestamp regression.

       @staticmethod
       async def notify_new_wish(db: AsyncSession, wish: Wish, submitter: User) -> None:
           # D-F04: fan out to role='chef' with feishu_open_id IS NOT NULL
           result = await db.execute(
               select(User.feishu_open_id).where(
                   User.role == "chef",

               )
           )
           receive_ids = list(result.scalars().all())
           payload = {
               "notification_type": "new",
               "wish_id": wish.id,
               "dish_name": wish.dish_name,
               "submitter_name": (submitter.display_name or submitter.username) if submitter else "未知用户",
               "reference_url": wish.reference_url,
               "note": wish.note,
           }
           from app.integrations.feishu import feishu_client
           for receive_id in receive_ids:
               try:
                   sent = await feishu_client.send_wish_notification(receive_id, payload)
                   if not sent:
                       print(f"⚠️ 愿望飞书通知发送失败: wish_id={wish.id}")
               except Exception as exc:
                   print(f"⚠️ 愿望飞书通知发送失败: wish_id={wish.id}, error={exc}")

       @staticmethod
       async def notify_claimed_wish_change(
           db: AsyncSession,
           wish: Wish,
           submitter: User,
           notification_type: str,   # "edit" | "cancel"
           claimed_by_chef_id: int,
           old_note: Optional[str],
       ) -> None:
           # D-F04: exactly the stored claimer (resolve by ID, not by role filter — A5 admin-claimer path)
           result = await db.execute(
               select(User.feishu_open_id).where(User.id == claimed_by_chef_id)
           )
           receive_id = result.scalar_one_or_none()
           if not receive_id:
               return  # chef without binding → silent skip
           submitter_name = (submitter.display_name or submitter.username) if submitter else "未知用户"
           action_verb = "修改了" if notification_type == "edit" else "撤回了"
           payload = {
               "notification_type": notification_type,
               "wish_id": wish.id,
               "dish_name": wish.dish_name,
               "change_description": f"{submitter_name} {action_verb}愿望: {wish.dish_name}",
               "old_note": old_note,
               "note_changed": (old_note != wish.note),
               "new_note": wish.note,
           }
           from app.integrations.feishu import feishu_client
           try:
               sent = await feishu_client.send_wish_notification(receive_id, payload)
               if not sent:
                   print(f"⚠️ 愿望飞书通知发送失败: wish_id={wish.id}")

               print(f"⚠️ 愿望飞书通知发送失败: wish_id={wish.id}, error={exc}")
