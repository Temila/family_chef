      ```js
      import { buildSync } from 'esbuild';
      import path from 'node:path';
      import { fileURLToPath } from 'node:url';
      const PLACEHOLDER = '<!-- fc-bootstrap -->';
      export default function inlineThemeBootstrap() {
        const entry = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '../src/theme/fouc-bootstrap.js');
        const bundle = () => buildSync({ entryPoints: [entry], bundle: true, format: 'iife', minify: true,

          name: 'inline-theme-bootstrap',
          transformIndexHtml(html) {

        };
      ```
